cmake_minimum_required(VERSION 3.28)
project(rambl_llama LANGUAGES C CXX)
option(RAMBL_CPU_AVX2 "Build the AVX2/FMA/F16C CPU variant" OFF)

# This bridge has no ggml DLL dependency: STT runtimes carry other ggml versions.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_AVX ${RAMBL_CPU_AVX2} CACHE BOOL "" FORCE)
set(GGML_AVX2 ${RAMBL_CPU_AVX2} CACHE BOOL "" FORCE)
set(GGML_FMA ${RAMBL_CPU_AVX2} CACHE BOOL "" FORCE)
set(GGML_F16C ${RAMBL_CPU_AVX2} CACHE BOOL "" FORCE)
set(GGML_BMI2 OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)

add_subdirectory("${LLAMA_SOURCE_DIR}" llama EXCLUDE_FROM_ALL)
add_library(rambl-llama SHARED rambl-llama.cpp)
if(RAMBL_CPU_AVX2)
    set_target_properties(rambl-llama PROPERTIES OUTPUT_NAME "rambl-llama-avx2")
endif()
target_compile_features(rambl-llama PRIVATE cxx_std_17)
target_compile_definitions(rambl-llama PRIVATE RAMBL_LLAMA_VERSION="${RAMBL_LLAMA_VERSION}")
target_link_libraries(rambl-llama PRIVATE llama-common mtmd)
